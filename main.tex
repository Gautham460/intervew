\section{Algorithms / Procedures}
\begin{frame}{Algorithm 1}{Interview Practice Flow}
\textbf{Setup Phase:}
\begin{enumerate}
    \item Candidate selects interview type (Company Q, Adaptive, Group, etc.)
    \item Select difficulty level: Easy, Medium, Hard
    \item Request camera/microphone permissions
    \item Initialize video recording infrastructure
\end{enumerate}

\vspace{3pt}
\textbf{Interview Phase:}
\begin{enumerate}
    \item Display interview question(s)
    \item Record candidate's answer (video + audio)
    \item Send to Google Gemini for analysis
    \item Generate AI feedback with 95th confidence
    \item Log performance metrics to Firestore
\end{enumerate}
\end{frame}

% ============================================================
% SLIDE 19: Algorithm - Adaptive Difficulty
% ============================================================
\begin{frame}{Algorithm 2}{Adaptive Difficulty Adjustment}
\begin{enumerate}
    \item Evaluate candidate's answer using Gemini AI
    \item Calculate performance score (0-100)
    \item Check difficulty adjustment rules:
    \begin{itemize}
        \item \textbf{Promote}: Score $\geq$ 80\% AND 2+ successful $\rightarrow$ Increase difficulty
        \item \textbf{Demote}: Score $<$ 50\% $\rightarrow$ Decrease difficulty
        \item \textbf{Maintain}: Keep current difficulty otherwise
    \end{itemize}
    \item Update user performance profile
    \item Calculate learning curve from historical data
    \item Retrieve next question at appropriate difficulty
\end{enumerate}
\end{frame}

% ============================================================
% SLIDE 20: Algorithm - AI Feedback
% ============================================================
\begin{frame}{Algorithm 3}{AI Feedback Generation}
\begin{enumerate}
    \item Extract candidate's answer text and video metadata
    \item Prepare prompt for Google Gemini with context:
    \begin{itemize}
        \item Interview question
        \item Expected answer (if available)
        \item Candidate's actual answer
        \item User's skill level and history
    \end{itemize}
    \item Call Gemini API with structured format request
    \item Parse response: score, strengths, improvements, hints
    \item Store feedback in Firestore with timestamp
    \item Calculate improvement trend
    \item Return formatted feedback to UI (<2 second SLA)
\end{enumerate}
\end{frame}

% ============================================================
% SLIDE 21: Algorithm - Proctoring
% ============================================================
\begin{frame}{Algorithm 4}{Proctored Exam Monitoring}
\textbf{Continuous Monitoring Loop:}
\begin{enumerate}
    \item Poll system events every 500ms during exam:
    \begin{itemize}
        \item Tab switch detection (blur/focus events)
        \item Keystrokes (Ctrl+C, Ctrl+V detection)
        \item Audio status (muted or unmuted)
        \item Face presence via camera
    \end{itemize}
    \item \textbf{Violation Detection}: If suspicious activity detected
    \begin{itemize}
        \item Record violation type, timestamp, details
        \item Increment violation counter
        \item Display warning to candidate (first violation)
        \item Terminate session if >3 violations
    \end{itemize}
    \item \textbf{Integrity Score}: Score = 100 - (violations $\times$ 10)
    \item \textbf{Report Generation}: Provide violation summary
\end{enumerate}
\end{frame}

% ============================================================
% SLIDE 22: Algorithm - Resume Analysis
% ============================================================
\begin{frame}{Algorithm 5}{Resume Analysis Processing}
\begin{enumerate}
    \item User uploads resume (PDF/DOC/TXT)
    \item Extract text content from file
    \item Parse resume structure: contact, summary, skills, experience
    \item Extract keywords using NLP (Google Gemini)
    \item Categorize skills: Technical, Leadership, Tools, Languages
    \item Generate ATS compatibility score:
    \begin{itemize}
        \item Check formatting (sections, length, clarity)
        \item Check keyword density (target vs actual)
        \item Check readability score
    \end{itemize}
    \item Compare against job description (if provided)
    \item Generate improvement suggestions
    \item Return formatted analysis with scores
\end{enumerate}
\end{frame}

% ============================================================
% SLIDE 23: Algorithm - Analytics
% ============================================================
\begin{frame}{Algorithm 6}{Analytics \& Performance Tracking}
\textbf{Performance Metrics Calculation:}
\begin{enumerate}
    \item Collect all interview records for user
    \item Calculate per-interview metrics:
    \begin{itemize}
        \item Answer quality score (from AI feedback)
        \item Clarity score (from video analysis)
        \item Confidence level (from tone/pace analysis)
    \end{itemize}
    \item Aggregate by skill (group questions by domain)
    \item Calculate trend analysis (improvement over time)
    \item Identify weak areas: scores $<$ 60\%
    \item Generate recommendations: Focus on weak areas
    \item Display analytics dashboard with visualizations
\end{enumerate}
\end{frame}
