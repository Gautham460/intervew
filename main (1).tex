\section{Introduction}
\begin{frame}{Introduction}{Project Overview}
\begin{itemize}
    \item Intervue is a \textbf{full-stack AI-powered mock interview platform} enabling candidates to practice technical and behavioural interviews and receive real-time intelligent feedback

    \item Designed as a \textbf{comprehensive career readiness suite} combining AI coaching, collaborative coding sessions, resume analysis, and enterprise hiring management

    \item Built on a \textbf{React + Vite frontend} with Firebase as the backend for authentication and data persistence, and Gemini AI for all intelligence features

    \item Supports \textbf{real-time collaborative coding} (Monaco Editor), proctored interview mode with eye-tracking, and group practice via shared sessions

    \item Integrates \textbf{AI Coach (Gemini-powered)} for adaptive hints and feedback, resume builder with PDF export, company-specific question banks, and study plan generation

    \item Features include mock interview sessions, analytics dashboard, video recording with playback, interview scheduling, community leaderboards, and enterprise candidate management
\end{itemize}
\end{frame}

% ============================================================
% SLIDE 3: Problem Statement
% ============================================================
\section{Problem Statement}
\begin{frame}{Problem Statement}

Candidates preparing for software engineering interviews lack access to \textbf{realistic, on-demand mock interview environments} that simulate actual technical rounds. Generic practice platforms provide no personalised feedback, no adaptive hints, and no performance history.

\vspace{8pt}

Existing tools are \textbf{siloed}: resume builders, coding platforms, and scheduling tools live in separate apps with no shared context. Candidates must manage multiple subscriptions and manually track their progress across platforms.

\vspace{8pt}

Employers and university career centres lack \textbf{visibility into bulk candidate readiness}. There is no unified enterprise dashboard to view candidate performance metrics, schedule interviews, or manage hiring pipelines -- all without building proprietary tooling.

\end{frame}

\section{Project Objectives}
\begin{frame}{Project Objectives}{What We Built}
\small
\begin{itemize}
    \item \textbf{AI Mock Interviews} -- Generate dynamic interviews from job role, tech stack \& experience; evaluate answers with Gemini AI and produce structured feedback reports

    \item \textbf{Collaborative Coding (Collab)} -- Real-time shared Monaco Editor sessions with language selection, code execution, and built-in video/audio via MediaDevices

    \item \textbf{AI Coach} -- Context-aware hints, answer evaluations, and progressive complexity adjustments using Gemini generative AI

    \item \textbf{Resume Builder \& Analyser} -- Drag-and-drop resume editor with live preview, PDF export (html2canvas + jsPDF), and AI-powered gap analysis

    \item \textbf{Proctored Mode} -- Eye-tracking overlay, tab-switch detection, and integrity monitoring during timed interview sessions

    \item \textbf{Enterprise Portal} -- Organisation-level analytics, candidate pipeline management, and secret-code-gated admin access via Firebase-backed role control

    \item \textbf{Deployed \& Secure} -- Clerk authentication with role-based routing, Firebase Firestore, production Vite build on Vercel
\end{itemize}
\end{frame}

\section{Algorithms / Procedures}

% SLIDE 20: Clerk Authentication
\begin{frame}{Algorithm 1}{Clerk Authentication \& Role Routing}
\textbf{Sign-Up:}
\begin{enumerate}
    \item User visits \texttt{/select-role} and selects Candidate or Enterprise
    \item Enterprise path: modal prompts for secret code; Firestore verifies
    \item If verified: \texttt{sessionStorage.setItem(``admin\_preauth'', ``true'')}
    \item Redirect to \texttt{/signin} with role state passed via React Router
    \item Clerk renders branded dark-theme sign-in; user completes registration
    \item On success: Clerk issues session token; React Router redirects to dashboard
\end{enumerate}

\vspace{3pt}
\textbf{Route Protection:}
\begin{enumerate}
    \item \texttt{ProtectRoutes} wraps all authenticated pages; redirects to \texttt{/signin} if no session
    \item \texttt{AdminRoute} additionally checks \texttt{user.publicMetadata.role === 'admin'} or \texttt{admin\_preauth}
    \item Redirect to \texttt{/select-role} with toast on denial
\end{enumerate}
\end{frame}

% SLIDE 21: Interview Generation
\begin{frame}{Algorithm 2}{AI Interview Generation \& Scoring}
\begin{enumerate}
    \item Candidate submits: \texttt{jobPosition}, \texttt{jobExperience}, \texttt{techStack}

    \item \textbf{Prompt Construction:} Structured JSON prompt sent to Gemini 1.5 Flash

    \item \textbf{Response Parsing:} Extract JSON array of \texttt{\{question, answer\}} objects

    \item \textbf{Persistence:} Write interview record to Firestore \texttt{interviews} collection

    \item \textbf{Answer Submission:} Each answer sent with question context to Gemini

    \item \textbf{Scoring Algorithm:}
    \begin{itemize}
        \item Gemini returns: \texttt{rating} (1--10), \texttt{feedback} (string), \texttt{correctAnswer}
        \item Frontend renders structured feedback card per question
    \end{itemize}

    \item \textbf{Persistence:} Write each \texttt{userAnswer} with score to Firestore

    \item \textbf{Analytics Update:} Recompute streak, average rating, badge eligibility
\end{enumerate}
\end{frame}

% SLIDE 22: Collaboration Code Execution
\begin{frame}{Algorithm 3}{Collaborative Code Execution (Browser-Safe)}
\begin{enumerate}
    \item Candidate writes JavaScript or TypeScript in Monaco Editor

    \item \textbf{Language Check:} Only JS/TS executed; others show ``unsupported'' message

    \item \textbf{Console Capture:} Override \texttt{console.log} to capture output to array

    \item \textbf{Execution:}
    \begin{itemize}
        \item Wrap code in try/catch
        \item Execute via \texttt{eval(code)} in isolated branch
        \item Restore \texttt{console.log} in \texttt{finally} block
    \end{itemize}

    \item \textbf{Output Rendering:} Write captured logs to terminal panel below editor

    \item \textbf{Error Handling:} Catch runtime errors; display \texttt{Error: \{message\}} in terminal

    \item \textbf{Camera/Mic:} \texttt{navigator.mediaDevices.getUserMedia} on toggle; stream attached to \texttt{<video>} ref; tracks stopped on toggle-off

    \item \textbf{Invite:} \texttt{navigator.clipboard.writeText(window.location.href)} on click
\end{enumerate}
\end{frame}

% SLIDE 23: Admin Access Control
\begin{frame}{Algorithm 4}{Enterprise Admin Access Control}
\textbf{Secret Code Verification Pipeline:}
\begin{enumerate}
    \item User clicks ``Enterprise'' card on \texttt{/select-role}
    \item Dialog modal opens with password input (max 6 chars)
    \item On submit: \texttt{getDoc(doc(db, ``admin\_codes'', ``admin1''))}
    \item Compare: \texttt{docSnap.data().code === enteredCode}
    \item If match: set \texttt{sessionStorage} flag; navigate to \texttt{/signin}
    \item If mismatch: show toast error; do not navigate
\end{enumerate}

% SLIDE 24: Proctored Mode
\begin{frame}{Algorithm 5}{Proctored Interview Mode}
\begin{enumerate}
    \item Candidate starts session in Proctored Mode

    \item \textbf{Camera Activation:} \texttt{getUserMedia} captures video feed for eye tracking

    \item \textbf{Tab-Switch Detection:}
    \begin{itemize}
        \item \texttt{document.addEventListener('visibilitychange', handler)}
        \item Increment violation counter; display warning overlay at threshold
    \end{itemize}

    \item \textbf{Eye-Tracking Overlay:} Face landmarks API (or heuristic gaze detection) flags looking away for $>$3 seconds

    \item \textbf{Integrity Timer:} Countdown per question; auto-advance on timeout

    \item \textbf{Report Generation:} At session end, compile: violations count, time-per-question, gaze scores

    \item Return success; frontend displays integrity report card
\end{enumerate}
\end{frame}

% SLIDE 25: Analytics Computation
\begin{frame}{Algorithm 6}{Analytics \& Badge Computation}
\textbf{Analytics Procedure:}
\begin{enumerate}
    \item Fetch all \texttt{userAnswers} for current user from Firestore
    \item Group by \texttt{mockIdRef} to compute per-interview average rating
    \item Compute overall average, best score, total sessions
    \item Calculate skill distribution from \texttt{techStack} fields
    \item Determine streak: consecutive days with completed sessions
\end{enumerate}

\vspace{3pt}
\textbf{Badge Triggers:}
\begin{itemize}
    \item First Interview Completed -- Authentication events
    \item 5-Day Streak -- Consecutive daily sessions
    \item Perfect Score -- Rating $\geq$ 9.5 on all questions
    \item Collab Master -- 3+ collaborative sessions completed
    \item Resume Pro -- Resume exported to PDF
\end{itemize}
\end{frame}
